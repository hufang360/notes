## 目录结构
```bash
tree ~/opt/tshock

~/opt/tshock
├── docker-compose.yml

# 测试用
├── tshock    # 服务器配置目录
├── plugins   # 插件目录
├── worlds    # 地图目录

# 第98个服
├── S98
│   ├── tshock         # 服务器配置目录
│   ├── plugins        # 插件目录
│   ├── world.wld      # 地图
│   ├── world.wld.bak  # 地图 备份1
│   └── world.wld.bak2 # 地图 备份2
    
# 第99个服
# 早期部署方法：用脚本启动linux版本
└── S99
    ├── bin
    ├── GeoIP.dat
    ├── i18n            # 语言文件
    ├── plugins         # 插件目录2，我一般不放插件
    ├── clear.sh        # 清理脚本
    ├── run.sh          # 启动脚本
    ├── ServerLog.txt
    ├── ServerPlugins   # 插件目录，插件我一般放这里
    ├── tshock          # 服务器配置目录
    ├── TShock.Server   # tshock 主可执行程序
    ├── world.wld       # 地图
    ├── world.wld.bak
    └── world.wld.bak2

```

## 目录映射

```yaml
# docker-compose.yml 部分截取

# 绝对路径
services:
  tshock:
    volumes:
      - /root/S1/tshock/:/tshock
      - /root/S1/plugins/:/plugins
      - /root/S1/worlds/:/worlds

# 相对路径
# 文件在 docker-compose.yml 同目录下）
services:
  tshock:
    volumes:
      - ./tshock/:/tshock
      - ./plugins/:/plugins
      - ./worlds/:/worlds

```


## 沿用
```yaml
# 地图
# 个人偏好，地图是直接放在S99目录下
# 文件名是 world.wld，注意没有s，不要写成 worlds.wld
# ./S99/world.wld -> 容器的 /worlds/world.wld
# tips: skyblock 大世界地图，也才30kb左右，测试时用这个创建地图也超快，图格数据超少，也更省内存

# 插件
# ./S99/ServerPlugins/ 目录下有 TShockAPI.dll
# 会和容器的/server/ServerPlugins/TShockAPI.dll 冲突
# 改名代替删除，注意此时要改扩展名，以防止加载，下回需要用原版开服再改回来
# 小tips：你更新新插件，也可以用改扩展名的方式，禁用旧插件
# mv ./S99/ServerPlugins/TShockAPI.dll ./S99/ServerPlugins/TShockAPI.dll.1

services:
  tshock:
    volumes:
      - ./S99/tshock/:/tshock
      - ./S99/ServerPlugins/:/plugins
      - ./S99/:/worlds
    command: >
      -lang 7
      -world /worlds/world.wld
      -autocreate 3
      -difficulty 2
      -seed "skyblock"

```


## 跑旧版
```yaml
# 使用linux版可执行程序，docker仅提供运行环境

# 插件
# 此时无需映射./S99/ServerPlugins
# 因为 容器的是/server/ServerPlugins
# 但需保证 ./S99/ServerPlugins/TShockAPI.dll 存在

# 经过测试，./S99/ 同时映射给 /worlds 和 /server，并不会冲突
# 记得 chmod +x ./S99/TShock.Server
services:
  tshock:
    volumes:
      - ./S99/tshock/:/tshock
      - ./S99/:/worlds
      - ./S99/:/server


# mac能用linux版可执行程序吗？
# 可以。
# 但mac默认用arm64的镜像，得指定成amd64。因为docker本身就是linux所以能跑。
services:
  tshock:
    image: ghcr.io/pryaxis/tshock:general-devel
    platform: linux/amd64
    volumes:
      - ./S99/tshock/:/tshock
      - ./S99/:/worlds
      - ./S99/:/server
        
# 容器内的运行时
# ghcr.io/pryaxis/tshock:6.1.0
# tshock 6.x  -> .NET9.0
# tshock 5.x  -> .NET6.0

```
